function mostCommonWord(string $text): string
{
    $counts = [];
    $best = "";
    foreach (explode(" ", strtolower($text)) as $word) {
        $counts[$word] = ($counts[$word] ?? 0) + 1;
        if ($counts[$word] > ($counts[$best] ?? 0)) {
            $best = $word;
        }
    }
    return "\"$best\" appears {$counts[$best]} times";
}
